Tentu! Berikut fungsi JavaScript untuk validasi email:

function validateEmail(email) {
  // Pola regex untuk validasi email
  const pattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  return pattern.test(email);
}

// Contoh penggunaan:
const email1 = "contoh@email.com";
const email2 = "email-tidak-valid";

console.log(validateEmail(email1)); // true
console.log(validateEmail(email2)); // false

Fungsi ini menggunakan regex untuk memeriksa format email. Anda juga bisa menambahkan validasi tambahan sesuai kebutuhan.
